# Lean requirements for the Streamlit Community Cloud deployment (app/).
# Deliberately does NOT include torch, sentence-transformers, pypdf, or
# langchain-text-splitters -- the deployed app loads a prebuilt,
# committed Chroma index (app/prebuilt_index/, built offline by
# app/build_index.py using the full torch/HuggingFace pipeline) and only
# ever embeds the user's query at runtime, via ONNX
# (app/onnx_embeddings.py), not the corpus. See the README's Phase 3
# section for the memory reasoning -- torch alone costs ~390MB RSS just
# to import; onnxruntime costs roughly a tenth of that. Every extra
# package here is memory the 1GB ceiling doesn't have to spare.
#
# Also does NOT include eval/dev dependencies (pytest, nbformat, pyyaml,
# langchain-classic) -- see requirements.txt / requirements-dev.txt for
# phase 1/2's fuller sets, which is what app/build_index.py and
# app/export_onnx_model.py need instead (torch-based, run offline, never
# part of the deployed app).

streamlit==1.59.2
langchain-core==1.4.9
langchain-chroma==1.1.0

# ONNX query-embedding path (app/onnx_embeddings.py) -- no torch, and
# deliberately `tokenizers` (Hugging Face's Rust tokenizer bindings)
# rather than `transformers.AutoTokenizer`: both read the same
# tokenizer.json and produce identical token ids, but importing the
# `transformers` package alone costs ~340MB RSS (model/config registry
# overhead, unrelated to torch) versus a few MB for `tokenizers` --
# confirmed by measurement, see README's Phase 3 section.
onnxruntime==1.27.0
tokenizers==0.22.2
numpy==2.4.6

# Optional LLM-generation arm (off by default; only exercised if a user
# pastes an API key in the sidebar -- see app/app_logic.py generate_answer).
# Lightweight to import; installed unconditionally so the feature works
# without a redeploy, but never required to have a valid key.
langchain-openai==1.3.5
